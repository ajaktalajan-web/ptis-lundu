        function submitMovement(e) {

            e.preventDefault();

            const staffName = document.getElementById('movStaffName').value;

            const category = document.getElementById('movCategory').value;

 

            const newMov = {

                id: Date.now(),

                date: selectedDate,

                staffName: staffName,

                category: category,

                slot: document.getElementById('movTimeSlot').value,

                destination: document.getElementById('movDestination').value.trim(),

                startTime: document.getElementById('movStartTime').value,

                endTime: document.getElementById('movEndTime').value

            };

 

            movements.unshift(newMov);

           

            const targetStaff = staffList.find(s => s.name === staffName);

            if (targetStaff) {

                if (!historicalAttendance[selectedDate]) historicalAttendance[selectedDate] = {};

                historicalAttendance[selectedDate][targetStaff.id] = category;

                localStorage.setItem('ptis_historical_attendance', JSON.stringify(historicalAttendance));

               

                // Auto Sync ke Google Sheets

                syncToGoogleSheets(targetStaff.id, targetStaff.name, category, selectedDate);

                loadStaffList();

            }

 

            loadMovements();

            updateStatistics();

            closeMovementModal();

            document.getElementById('movementForm').reset();

            alert(`Pergerakan keluar pegawai bagi tarikh (${selectedDate}) berjaya direkodkan!`);

        }

 

        function updateStatistics() {

            const total = staffList.length;

            let presentCount = 0;

           

            staffList.forEach(s => {

                const st = getStaffStatusForDate(s.id);

                if (st === 'Di Pejabat') presentCount++;

            });

 

            const percent = total > 0 ? Math.round((presentCount / total) * 100) : 0;

            const dateMovements = movements.filter(m => !m.date || m.date === selectedDate);

 

            document.getElementById('statTotalStaff').innerText = total;

            document.getElementById('statPresentCount').innerText = presentCount;

            document.getElementById('statPresentPercent').innerText = `${percent}% Berada di Pejabat`;

            document.getElementById('statOutCount').innerText = dateMovements.length;

        }
